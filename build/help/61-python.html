The packages in Python

These are the main functions of each package, for use in notebooks and scripts. You can add each box to the open notebook. The packages' own help, such as help(studio2ga4gh.convert_file), tells you more.

studio2ga4gh, ga4gh2studio

import json
from studio2ga4gh import convert_file, parse_session

# a drawing as Python objects: people, couples, pedigree numbers, the colour key
drawing = parse_session(json.load(open("/data/examples/okafor.studio.json")))
print(drawing.legend, drawing.numbers())

# the drawing → the two GA4GH files (terms: a terms file, as a dict)
terms = json.load(open("/data/examples/example-terms.json"))
result = convert_file("/data/examples/okafor.studio.json", "okafor-py", "/home/user/results", terms=terms)
print(result.summary, result.warnings)

# and back: a GA4GH file → a drawing
from ga4gh2studio import convert_file as draw
print(draw("/home/user/results/okafor-py_ga4gh-pedigree.json", None, "/home/user/pedigrees/okafor-py.studio.json").summary)

pedigree_ga4gh: reading, checking, upgrading

import pedigree_ga4gh as pg

doc = pg.load("/data/examples/okafor_ga4gh-pedigree.json")
findings = pg.validate(doc)                       # [] when the file is correct
print(pg.render_findings(findings), pg.is_valid(findings))

# any GA4GH form → a Kindred: members, their terms, the KIN relations between them
kindred, notes = pg.from_document(doc)
for m in kindred.members[:3]:
    print(m.id, m.sex, [t.label for t in m.terms])
print(kindred.genetic_parents("IV-3"))              # (father, mother, notes)

# and a Kindred → either form
family, warnings = pg.to_phenopackets_family(kindred, created_by="my notebook")

# an older file, rewritten in the standard form
old = pg.load("/data/examples/older/abusua2ga4gh-0.2_phenopackets-family.json")
new, changes, todo = pg.upgrade(old)
print(len(changes), "changes;", todo)

phenopacket_toolkit: what phenotk does

import phenopacket_toolkit as ptk
from phenopacket_toolkit.ops import render, stats, render_stats, filter_document, deidentify, to_kind, to_format

doc = ptk.read("/data/examples/okafor_ga4gh-pedigree.json")    # a Document, whatever its kind
print(doc.kind, render(doc))
print(render_stats(stats(doc)))

affected = filter_document(doc, affected_only=True)
family, notes = to_kind(doc, "family")
print(to_format(family, "ped"))

shared = deidentify(doc, remap_ids="seq")
ptk.write(shared.document, "/home/user/results/okafor-py.deidentified.json")

The other operations – normalise, diff, split, merge, add and sort – are in phenopacket_toolkit.ops too.

hponorm, mondonorm: searching the ontologies

from mondonorm.mondo import MondoIndex      # a copy of Mondo among your files (or: from mondonorm.ols import OlsClient)
index = MondoIndex.load("/home/user/ontologies/mondo.obo", allow_download=False)
for c in index.search("bowel cancer", limit=5):
    print(c.id, c.label, c.score, c.matched_on)

hponorm.hpo.HpoIndex does the same for the HPO. OlsClient().search(…) searches EBI's Ontology Lookup Service over the internet instead.

Many families at once

import glob, json, os
from studio2ga4gh import convert_file
import pedigree_ga4gh as pg

terms = json.load(open("/data/examples/example-terms.json"))
os.makedirs("/home/user/results/all", exist_ok=True)
for path in sorted(glob.glob("/data/examples/*.studio.json")):
    name = os.path.basename(path).replace(".studio.json", "")
    result = convert_file(path, name, "/home/user/results/all", terms=terms, family_id=name.upper())
    for out, _ in result.files:
        print(os.path.basename(out), "valid" if pg.is_valid(pg.validate(pg.load(out))) else "NOT valid")
